# Самые часто используемые команды
 
```bash
# клонирование репозитория
git clone <url-адрес репозитория>

# показать состояние репозитория
git status      

# дабавить файлы в индекс
git add <file1> <file2> ... <fileN>  

# добавить все файлы в папке
git add .  

# убрать из индекса
git rm --cached <file|dir>

# зафиксиовать изменения
git commit –m "<Введите ваше сообщение о фиксации здесь>"

# Перезаписать последний коммит
git commit --amend

# просмотр локальных веток
git branch

# просмотр и локальных и удалённых веток
git branch -a

# создать локально новую ветку, но остаться на прежней
git branch <название ветки>

# переключиться на другую ветку
git checkout <название ветки>

# git branch + git checkout
git checkout -b <название ветки>

# удалить ветку
git branch -d <название ветки>

# удалить ветку принудительно
git branch -D <название ветки>

# удалить удаленную ветку
git push origin --delete <название ветки>

# слияние веток (сначала переключаемся)
git checkout <ветку, в кот. нужно залить изменения>
git merge <ветка, из кот. заливаем изменения>
# при конфликте, чтобы посмотреть какие файлы не объединены
git status

# отменить слияние
git merge --abort

# извлечь изменения из удал. репозитория
# git fetch + git merge
git pull

# извлечь и просмотреть изменения
git pull --verbose

# отправить изменения
git push

# отправить новую ветку
git push -u origin new_branch

# удалить удаленную ветку
git push --delete origin existing_branch

# показать список коммитов (p - подробно)
git log -p

# просмотр изменений отдельного коммита
git show <хэш коммита (можно кратко)>

# просмотр изменений до коммита
git diff

# разница между индексом и последним коммитом
git diff --staged

# разница между 2-мя коммитами
git diff <хэш 1-го коммита> <хэш 2-го>

# удалить файлы каталога и индекса
git rm <файл>

# отмена всех изменний до индексации
git checkout -f

# отмена изменений в одном файле
git checkout <файл>

# отменить индексацию файла
git reset HEAD <файл>

# временно сбросить незафикс. измен-я
git stash
# вернуть сброшен. измен-я
git stash pop 

# откат последнего коммита (создаст новый коммит)
git revert HEAD

# откат до указанного коммита
git revert <хэш коммита (можно кратко)>

# слияние, с другим порядком коммитов в истории:
# 1-й коммит нашей ветки становится после 
# последнего коммита другой ветки
git rebase branch_name

# добавление удалённого репозитория
git remote add awesomeapp https://github.com/someurl..

# просмотр удалённых URL-адресов
git remote -v

# бинарный поиск бага в истории
git bisect

# посмотреть авторов изменений
git blame

# поиск по регул. выражению в истории гита
git grep

# посмотреть где находился указатель HEAD в любой момент времени
git reflog
```
