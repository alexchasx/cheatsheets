
# Работа с паролями в Linux

```bash
# файлу(или папке) присваиваем нового владельца:
sudo chown root:root путь/к/файлу/или/папке

# оставляем права читать и записывать в файл только владельцу(он же root)
sudo chmod 600 путь/к/файлу/или/папке

# пример записи в файл c именем 12
cat << wer >> 12
testsdfsdfsd
asasdasda
wer
```
