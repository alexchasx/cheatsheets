
# Работа с файлами в PHP (конспект).

```php
// 1. Чтение файла

// Абсолюттный путь
$path = __DIR__ . '/data.txt';

// Если существует файл/каталог и доступен для чтения
if (is_readable($path)) {

// 1) Чтение построчно:

    // Открывает файл и возращает его дескриптор
    // 'r'  - чтение
    // 'r+' - чтение и запись
    // 'w'  - запись или создать файл
    // 'w+' - запись и чтение или создать файл
    // 'a'  - запись в конец файла или создать файл
    $fh = fopen($path, 'r');    

    // Пока не достигнут конец файла
    while (!feof($fh)) {
        // Возращает одну строку (до символа перевода строки) 
        // и переводит курсор в файле на след. строку
        $line = fgets($fh);

        echo $line;
    }
    // Закрывает открытый дескриптор файла (снимает блокировку)
    fclose($fh);


// 2) Чтение файла целиком, помещая содержимое в массив
    $lines = file($path, FILE_IGNORE_NEW_LINES)


// 3) Чтение файла целиком в строку
    $lines = file_get_contents($path);


// 4) Передает содержимое в поток вывода
    readfile($path);
}


// 2. Запись в файл

// Если существует файл/каталог и доступен для записи

    $fh = fopen($path, 'w');
```
